### 👂 阶段 1：听觉层升级 (集成 ASR)

**目标**：让 V14.1 系统在输出声纹ID的同时，输出文字内容。

#### 步骤 1.1：编写 Whisper 服务类
创建文件 `backend/modules/audio_analysis/whisper_svc.py`：

```python
from faster_whisper import WhisperModel
import os

class WhisperService:
    def __init__(self, model_path="models/asr/faster-whisper-large-v3", device="cuda"):
        # compute_type="float16" 在 V100 上效率最高
        self.model = WhisperModel(model_path, device=device, compute_type="float16")
        print("✅ ASR Model Loaded")

    def transcribe(self, audio_array):
        """
        Args:
            audio_array: numpy array (16k sample rate)
        Returns:
            text: str
        """
        # beam_size=1 牺牲一点精度换取极致速度，适合实时流
        segments, _ = self.model.transcribe(audio_array, language="zh", beam_size=1)
        return "".join([s.text for s in segments])
```

#### 步骤 1.2：改造推理流水线
修改 `backend/pipeline/inference_pipeline.py`。这是最关键的一步，将 ASR 缝合进现有的 VAD 循环中。

```python
# ... 原有 import ...
from backend.modules.audio_analysis.whisper_svc import WhisperService # 新增

class InferencePipeline:
    def __init__(self, config):
        # ... 原有初始化 ...
        self.speaker_model = ...
        
        # [新增] 初始化 ASR
        self.asr_service = WhisperService()

    def _recognize_vad_batch(self, audio_tensor, sample_rate):
        # 1. VAD 切分 (保持原样)
        speech_segments = self.vad_model(audio_tensor)
        results = []

        for segment in speech_segments:
            # 获取音频片段 (Tensor 转 Numpy 给 Whisper 用)
            audio_chunk_tensor = audio_tensor[segment.start:segment.end]
            audio_chunk_numpy = audio_chunk_tensor.numpy()

            # --- 并行任务 A: 声纹 (原有) ---
            speaker_id, score = self._identify_speaker(audio_chunk_tensor)

            # --- 并行任务 B: ASR (新增) ---
            text_content = self.asr_service.transcribe(audio_chunk_numpy)

            # 3. 聚合结果
            results.append({
                "start_ts": segment.start_ts, # 绝对时间戳
                "end_ts": segment.end_ts,
                "speaker_id": speaker_id,
                "text": text_content,
                "score": score
            })
            
        return results
```

### 💾 阶段 2：数据层打通 (IoTDB)

**目标**：将听觉结果写入 IoTDB，以便与 P4 产生的视觉数据对齐。

#### 步骤 2.1：编写 IoTDB 连接器
创建文件 `backend/data/iotdb/connector.py`：

```python
from iotdb.Session import Session
from iotdb.utils.IoTDBConstants import TSDataType, TSEncoding, Compressor

class IoTDBConnector:
    def __init__(self, host="127.0.0.1", port=6667):
        self.session = Session(host, port, "root", "root")
        self.session.open(False)

    def insert_audio_event(self, device_id, timestamp, speaker, text):
        """
        写入音频事件
        Path: root.store_01.audio.{device_id}
        """
        device_path = f"root.store_01.audio.{device_id}"
        measurements = ["speaker", "text"]
        types = [TSDataType.TEXT, TSDataType.TEXT]
        values = [speaker, text]
        
        self.session.insert_record(device_path, timestamp, measurements, types, values)

    def query_visual_context(self, start_ts, end_ts):
        """
        查询视觉上下文 (给融合引擎用)
        Path: root.store_01.visual (假设 P4 已经往这里写数据了)
        """
        # 向前向后各扩展 5秒
        q_start = start_ts - 5000
        q_end = end_ts + 5000
        
        sql = f"SELECT class FROM root.store_01.visual WHERE time >= {q_start} AND time <= {q_end}"
        with self.session.execute_query_statement(sql) as session_data_set:
            # 简单统计逻辑：统计出现了哪些物体
            objects = []
            while session_data_set.has_next():
                row = session_data_set.next()
                objects.append(row.get_fields()[0]) # 假设第一列是 class
            return list(set(objects)) # 去重返回，如 ['customer', 'menu']
```

### 🧠 阶段 3：融合引擎开发 (LLM)

**目标**：让 V100 运行大模型，并根据 Prompt 进行判断。

#### 步骤 3.1：编写 LLM 客户端
创建文件 `backend/modules/llm/client.py`：

```python
from vllm import LLM, SamplingParams

class LLMClient:
    def __init__(self, model_path="models/llm/Qwen2.5-7B-Instruct-GPTQ-Int4"):
        # ⚠️ 关键：限制显存占用为 40%，给 ASR 和 声纹 留空间
        self.llm = LLM(model=model_path, gpu_memory_utilization=0.4, trust_remote_code=True)
        self.sampling_params = SamplingParams(temperature=0.1, max_tokens=256)
        print("✅ LLM Loaded")

    def analyze(self, prompt):
        # 构造 Qwen 的 Chat 格式
        messages = [
            {"role": "system", "content": "你是一个门店服务质量分析员。请根据输入信息判断员工行为。只输出JSON格式结果。"},
            {"role": "user", "content": prompt}
        ]
        # 这里简化处理，实际建议用 tokenizer.apply_chat_template
        # 为了演示直接拼接
        full_prompt = f"<|im_start|>system\n{messages[0]['content']}<|im_end|>\n<|im_start|>user\n{messages[1]['content']}<|im_end|>\n<|im_start|>assistant\n"
        
        outputs = self.llm.generate([full_prompt], self.sampling_params)
        return outputs[0].outputs[0].text
```

#### 步骤 3.2：编写融合控制器

from backend.modules.llm.client import LLMClient
from backend.data.iotdb.connector import IoTDBConnector

class FusionController:
    def __init__(self):
        self.iotdb = IoTDBConnector()
        self.llm = LLMClient()

    def process_event(self, audio_result):
        """
        audio_result: 阶段1流水线输出的字典
        """
        # 1. 获取视觉上下文
        visual_tags = self.iotdb.query_visual_context(
            audio_result['start_ts'], 
            audio_result['end_ts']
        )
        
        # 2. 构建 Prompt
        prompt = f"""
        [听觉信息]
        说话人: {audio_result['speaker_id']}
        内容: "{audio_result['text']}"
        
        [视觉信息]
        同时段画面中出现的物体: {', '.join(visual_tags)}
        
        [任务]
        判断员工是否在向顾客介绍商品？
        如果视觉中有'menu'或'goods'且语音包含介绍性词汇，则为True。
        
        [输出格式]
        {{"is_promotion": true/false, "reason": "..."}}
        """
        
        # 3. LLM 推理
        analysis_json = self.llm.analyze(prompt)
        
        # 4. 打印或存入 MySQL
        print(f"🤖 融合分析结果: {analysis_json}")
